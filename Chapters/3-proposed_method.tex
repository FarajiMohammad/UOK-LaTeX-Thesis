\section{فضای نهان مشترک}\label{lowrank}
ماتریس ویژگی به‌صورت $\bm{X} \in \mathbb{R}^{n \times d}$ تعریف می‌شود و ماتریس برچسب به‌صورت $\bm{Y} = \{\bm{y}_1, {\dots}, \bm{y}_l\} \subseteq \{0,1\} \in \mathbb{R}^{n \times l}$ تعریف می‌شود که
${Y}_{i,j}=1$
اگر نمونه $i$-ام، برچسب $j$-ام را داشته باشد، در غیر این‌صورت  ${Y}_{ij}=0$. در مسائل داده‌های چندبرچسبه، برچسب‌ها با یکدیگر مرتبط هستند، از این رو می‌توان از رتبه‌پایین این ماتریس استفاده کرد. محتویات ماتریس خلوت $\bm{Y}$، دارای مقادیر دودویی و ابعاد بالا می‌باشد، بنابراین یادگیری براساس این ماتریس مشکل است. لذا اندازه ماتریس کم‌بُعد $\bm{Y}$ بصورت $‎‎‎k<\text{‎\lr{min}‎}(n,l)$ خواهد بود،
از این رو، ماتریس برچسب را به دو ماتریس کوچکتر به‌صورت زیر تجزیه می‌کنیم:
\begin{align}
	\bm{Y}\simeq\bm{VB}
\end{align}
که $\bm{V} \in \mathbb{R}^{n \times k} $ و $\bm{B} \in \mathbb{R}^{k \times l}$.‎


$\bm{V}$
ماتریس نهان برچسب‌ها است که فشرده‌تر\LTRfootnote{Compact} و از نظر معنایی انتزاعی‌تر از ماتریس برچسب‌های اصلی هستند.
$\bm{B}$
نشان می‌دهد که چگونه برچسب‌های اصلی با برچسب‌های نهفته مرتبط هستند. اطلاعات مهم ماتریس برچسب اصلی $\bm{Y}$ در ماتریس کم‌بُعد $\bm{V}$ کدگذاری شده است که همچنین داده‌های نامطلوب ماتریس برچسب را کاهش می‌دهد. ماتریس های $\bm{V}$ و $\bm{B}$ با به حداقل رساندن خطای بازنمایی برچسب $\lVert\bm{Y} - \bm{VB}\rVert^2_F$ بدست می‌آید. بازنمایی برچسب نهان، با $\bm{V}$ نشان داده شده است، یک ماتریس با ابعاد کم و با مقدار حقیقی که حاوی اطلاعات متراکم می‌باشد. بنابراین یادگیری نگاشت پیوسته از فضای ویژگی به فضای برچسب نهفته نسبتاً آسان‌تر از فضای برچسب اصلی است\cite{zhu2017multi}.
یک فرض مهم در رابطه‌ی همبستگی بین ویژگی‌ها و برچسب‌ها این می‌باشد که ویژگی‌های مشابه، برچسب‌های مشابهی دارند. در نتیجه، اطلاعات مشترک بین فضای ویژگی و فضای برچسب باید سازگار باشد. ما درنظر می‌گیریم $\bm{V}$ یک ماتریس مشترک بین ماتریس ویژگی و ماتریس برچسب باشد. ماتریس $\bm{W} \in \mathbb{R}^{d \times k}$ بروزرسانی می‌شود جهت نگاشت نمونه‌ها به فضای نهان. با کاهش خطای تابع بازنمایی ویژگی $\lVert\bm{V} - \bm{XW}\rVert^2_F$، ماتریس وزن ویژگی $\bm{W}$ آموخته خواهد شد و با ادغام بازنمایی برچسب و خطای بازنمایی ویژگی در یک چارچوب ماتریس نامنفی \cite{lee1999learning}، فضای نهان مشترک را از طریق تابع زیر بدست می‌آوریم:
\begin{align}
	\min_{\bm{V},\bm{B},\bm{W}}&\lVert\bm{Y}-\bm{V}\bm{B}\rVert^2_F+\lVert\bm{V}-\bm{X}\bm{W}\rVert^2_F,\quad{\text{\lr{s.t.}}\quad}\bm{{V,B,W}}\geq 0
\end{align}
\section{حفظ ساختار محلی}\label{LS}
در بسیاری از کاربرد‌های یادگیری ماشین، فضای ویژگی می‌تواند دارای ابعاد بالا همراه با نویز باشد، که استخراج اطلاعات معنی‌دار از داده‌ها را به چالش می‌کشاند. یکی از روش‌های رایج برای استخراج اطلاعات مفید، تبدیل فضای ویژگی اصلی به فضای نهان با ابعاد پایین‌تر است، جایی که ساختار زیربنایی داده‌ها را می‌توان راحت‌تر بدست ‌آورد. با این حال، بازنمایی داده‌ها در فضایی با ابعاد پایین‌تر می‌تواند منجر به از دست رفتن اطلاعات\LTRfootnote{Loss of information} شود و ممکن است روابط پیچیده بین ویژگی‌ها را به درستی نشان ندهد، از این رو از  یک منظم‌ساز گراف \cite{cai2010graph} برای ارائه یک ماتریس مناسب $\bm{V}$ که انسجام بین فضای ویژگی اولیه و فضای ساختار نهان را تضمین ‌کند استفاده می‌کنیم. با توجه به اصل اساسی این قاعده‌بندی، اگر دو نمونه در فضای ماتریس ویژگی $\bm{X}$ با یکدیگر همبستگی داشته باشند، در فضای ساختار نهان نیز هر دو نمونه $\bm{v}_i$ و $\bm{v}_j$ با یکدیگر همبستگی خواهند داشت. به‌طور خاص، ما از یک منظم‌ساز گراف برای تشویق حفظ ساختار محلی در فضای نهان استفاده می‌کنیم، و اطمینان حاصل می‌کنیم که متغیرهای نهفته به‌طور دقیق ساختار ویژگی‌های اصلی را منعکس می‌کنند. فرمول منظم‌ساز گراف را می‌توان به‌صورت زیر بیان کرد:
\begin{align} \label{eq:mfold}
	\min_{\bm{V}}&\frac{1}{2}\sum_{i=1}^n\sum_{j=1}^n\lVert\bm{v}_i-\bm{v}_j\rVert^2S_{i,j}\nonumber\\
	&=\frac{1}{2}\sum_{i=1}^n\sum_{j=1}^n(\bm{v}_i^\top\bm{v}_iS_{i,j}-2\bm{v}_i^\top\bm{v}_jS_{i,j}+\bm{v}_j^\top\bm{v}_jS_{i,j})\nonumber\\
	&=\frac{1}{2}\sum_{i=1}^n\sum_{j=1}^n(2\bm{v}_i^\top\bm{v}_iS_{i,j}-2\bm{v}_i^\top\bm{v}_jS_{i,j})\nonumber\\
	&=\sum_{i=1}^n\bm{v}_i^\top\bm{v}_iD_{i,i}-\sum_{i=1}^n\sum_{j=1}^n\bm{v}_i^\top\bm{v}_jS_{i,j}\nonumber\\
	&=\mathrm{Tr}(\bm{V}^\top\bm{DV})-\mathrm{Tr}(\bm{V}^\top\bm{SV})=\mathrm{Tr}(\bm{V}^\top \bm{LV}),
\end{align}
که
$\bm{D}$ ماتریس قطری\LTRfootnote{Diagonal} و $\bm{S}$ ماتریس همبستگی متقارن\LTRfootnote{Symmetric} است. $\bm{L} = \bm{D} - \bm{S}$ ماتریس لاپلاسین گراف می‌باشد. با ادغام عبارات فوق در مدل، تابع زیر را بدست می‌آوریم:

\begin{align}\label{eq:m-mfold}
	\min_{\bm{V},\bm{B},\bm{W}}&\lVert\bm{Y}-\bm{V}\bm{B}\rVert^2_F+\lVert\bm{V}-\bm{X}\bm{W}\rVert^2_F +\lambda_1{\mathrm{Tr}\bm{({V}^\top{LV})}}
	,\nonumber \\
	&{\text{\lr{s.t.}}}\bm{V},\bm{B},\bm{W}\geq 0
\end{align}
که 
$\lambda_1$
پارامتر حفظ ساختار محلی است.
\section{همبستگی برچسب سراسری و محلی}\label{GLOCAL}
برای استفاده مؤثر از اطلاعات برچسب‌ها، باید از همبستگی‌های برچسب استفاده شود. در این روش، مدل پیشنهادی را با استفاده از همبستگی برچسب مرتبه‌بالا منظم می‌کنیم. در این راستا باید به همزیستی\LTRfootnote{Coexistence} همبستگی‌های سراسری و محلی برچسب اشاره کرد. برای ترکیب هر دوی آن‌ها، منظم‌ساز خمینه برچسب‌ها را در این قسمت معرفی می‌کنیم. مفهوم منظم‌ساز خمینه سراسری برچسب‌ ‌از منظم‌ساز خمینه سطح نمونه \eqref{eq:mfold} مشتق شده است. به‌طور دقیق‌تر اگر دو برچسب دارای همبستگی بالایی باشند، خروجی‌های طبقه‌بند مربوطه آن‌ها نیز باید شباهت داشته باشند و برعکس، برچسب‌هایی که دارای همبستگی کمتری هستند باید خروجی‌های طبقه‌بند آن‌ها شباهت کمتری داشته باشند. به عبارت دیگر، همبستگی‌های برچسب منجر به خروجی‌های طبقه‌بند مشابه می‌شود. در این تحقیق از شباهت کسینوس\LTRfootnote{Cosine Similarity} برای تعیین کمیت همبستگی سراسری برچسب استفاده می‌شود $\bm{C} \in\mathbb{R}^{l \times l}$، که به‌صورت زیر محاسبه می‌شود: 
\begin{align}\label{eq:gloCor}	
	{{{C}_{ij}=\frac{\bm{y}^\top_{i}{\bm{y}_{j}}}{{\lVert{\bm{y}_i}\lVert} \lVert{\bm{y}_{j}}\lVert}}}
\end{align}
که
$\bm{y}_i$ 
, 
$\bm{y}_j$
به مفهوم $i$-ام و $j$-ام بردار برچسب برای همه‌ی نمونه‌ها می‌باشد.‎\\
در\eqref{eq:m-mfold}، برچسب پیش‌بینی شده برای $\bm{x}$ به‌صورت $f(\bm{x})$ است، که $f(\bm{x}) = \bm{x}\bm{WB}$. 
$\bm{f} = \{{f}_1, {\dots}, {f}_l\}$
که $\bm{f}_j(\bm{x})$ یعنی $j$-ام برچسب برای نمونه $\bm{x}$ پیش‌بینی شده است. در نهایت پیش‌بینی برای همه‌ی ${n}$ نمونه در ماتریس پیش‌بینی $\bm{F}\in\mathbb{R}^{n \times l}$ ذخیره می‌شود،‌ $\bm{F}= \{f({\bm{x}}_1),{\dots},f({\bm{x}}_n)\}^\top = \bm{XWB}$ شامل محتویات پیش‌بینی شده برای $i$-ام برچسب $\bm{f}_i$، اگر برچسب ${i}$-ام و ${j}$-ام شبیه باشند باید بردار پیش‌بینی شده
$\bm{f}_j$ و $\bm{f}_i$
این دو برچسب نیز شبیه باشند و برعکس. تعریف منظم‌ساز خمینه ‌سراسری برچسب شبیه تعریف منظم‌ساز خمینه نمونه \eqref{eq:mfold} است و به‌صورت زیر تعریف می‌شود:
\begin{align}\label{eq:glomfold}
	\min_{\bm{F}}&\frac{1}{2}\sum_{i=1}^l\sum_{j=1}^l\lVert\bm{f}_i-\bm{f}_j\rVert^2C_{i,j}=\nonumber\\
	&\mathrm{Tr}(\bm{FA}\bm{F}^\top)-\mathrm{Tr}(\bm{FC}\bm{F}^\top)=\mathrm{Tr}(\bm{FP}\bm{F}^\top),\nonumber\\
\end{align}
$\bm{A}$ ماتریس قطری و به‌صورت ${A}_{ii}=\sum_{j=1}^l {C}_{ij}$ تعریف می‌شود. 
با کمینه‌سازی کردن \eqref{eq:glomfold}، خطای $\lVert\bm{f}_i-\bm{f}_j\rVert^2 $ کم خواهد بود. منظم‌ساز خمینه در \eqref{eq:glomfold} به‌صورت $\mathrm{Tr}(\bm{FPF}^\top)$، که $\bm{P} = \bm{A}-\bm{C}$ ماتریس لاپلاسین ‌است. $\bm{F}=\bm{XWB}$ برچسب‌های پیش‌بینی شده برای ویژگی‌ها می‌باشد. از آنجایی که همبستگی‌های برچسب می‌تواند در نواحی مختلف محلی متفاوت باشد، برای استخراج این همبستگی‌ها از منظم‌ساز خمینه محلی استفاده می‌کنیم. فرض می‌کنیم که مجموعه‌داده\LTRfootnote{Dataset} $\bm{X}$ به ${g}$ گروه $\{\bm{X}_1,{\dots},\bm{X}_g\}$ تقسیم می‌شود، ماتریس $\bm{X}_m \in \mathbb{R}^{n_m \times d}$ نمونه دارد. با استفاده از خوشه‌بندی، مانند شبکه‌ها\LTRfootnote{Networks} و مسیرهای ژنی\LTRfootnote{Pathways} \cite{chuang2007network,subramanian2005gene} در کاربردهای بیوانفورماتیک\LTRfootnote{Bioinformatics}، امکان دستیابی به این پارتیشن‌بندی وجود دارد. فرض  $\bm{C}_m \in \mathbb{R}^{{l} \times {l}}$ ماتریس همبستگی برچسب محلی یک گروه ${m}$ است. و $\bm{Y}_m$ زیر ماتریس برچسب در $\bm{Y}$ متناظر با $\bm{X}_m$ است. مانند \eqref{eq:gloCor}، ما همبستگی‌های برچسب محلی را به‌صورت زیر محاسبه می‌کنیم:

\begin{align}\label{eq:llc}	
	{{{C}^{(m)}_{i,j}}=
		\frac
		{{\bm{y}^{(m)}_{i}}^\top\bm{y}^{(m)}_{j}}
		{{\lVert{\bm{y}^{(m)}_{i}}\lVert}\lVert{\bm{y}^{(m)}_{j}}\lVert}}, \; m \in  \{1,{\dots},{g}\}
\end{align}
مشابه همبستگی برچسب سراسری \eqref{eq:glomfold}، اگر دو برچسب با همدیگر همبستگی داشته باشند باید بردار ‎‎پیش‌بینی شده‎ طبقه‌بند نیز شبیه باشند:
\begin{align}\label{eq:loclmfold}
	\min_{\bm{F}}&\sum_{m=1}^{g}\frac{n_m}{n}\sum_{i=1}^{l}\sum_{j=1}^{l} \lVert{\bm{f}^{(m)}_i}-{\bm{f}^{(m)}_j}\rVert^2 {C^{(m)}_{i,j}}\nonumber \\
	&=\sum_{m=1}^{g} \frac{n_m}{n}[\mathrm{Tr}(\bm{F}_m\bm{A}_m\bm{F}_m^\top) - \mathrm{Tr}(\bm{F}_m \bm{C}_m \bm{F}_m^\top)]\nonumber \\
	&= \sum_{m=1}^{g}\frac{n_m}{n}\mathrm{Tr}(\bm{F}_m \bm{P}_m \bm{F}_m^\top)
\end{align}
که $\bm{F}_m = \bm{X}_m\bm{WB}$ ماتریس خروجی طبقه‌بند برای گروه $m$ است و $\bm{P}_m$ ماتریس لاپلاسین $\bm{C}_m$ است. برای پوشش عدم تعادل\LTRfootnote{Imbalance} خوشه، منظم‌ساز همبستگی محلی برچسب را با ضریب ${n_m}/{n}$  نرمال می‌کنیم.
پس از افزودن منظم‌سازهای خمینه سراسری \eqref{eq:glomfold} و محلی  \eqref{eq:loclmfold} به تابع \eqref{eq:m-mfold}، فرمول زیر بدست می‌آید:
\begin{align}
	&\min_{\bm{V},\bm{B},\bm{W}}\lVert\bm{Y}-\bm{V}\bm{B}\rVert^2_F+\lVert\bm{V}-\bm{X}\bm{W}\rVert^2_F+\lambda_1{\mathrm{Tr}\bm{({V}^\top{LV})}}
	\nonumber \\&+\lambda_2[\mathrm{Tr}(\bm{FP}\bm{F}^\top) 
	+ \sum_{m=1}^{g}\frac{n_m}{n}\mathrm{Tr}(\bm{F}_m\bm{P}_m\bm{F}^\top_m) ],
	\quad{\text{\lr{s.t.}}\quad}\bm{V},\bm{B},\bm{W}\geq 0 
\end{align}
که $\lambda_2$ نشان دهنده تأثیر همبستگی برچسب سراسری و محلی برای تابع هدف است. در نهایت، تابع ما از نُرم $\ell_{2,1} $، که برای انتخاب ویژگی است استفاده می‌کند. در نتیجه تابع نهایی به‌صورت زیر تنظیم می‌شود: 
\begin{align}\label{eq:BeOpt}
	&\min_{\bm{V},\bm{B},\bm{W}}\lVert\bm{Y}-\bm{V}\bm{B}\rVert^2_F+\lVert\bm{V}-\bm{X}\bm{W}\rVert^2_F+\lambda_1{\mathrm{Tr}\bm{({V}^\top{LV})}} \nonumber \\ 
	& +\lambda_2[\mathrm{Tr}(\bm{FP}\bm{F}^\top)+\sum_{m=1}^{g}\frac{n_m}{n}\mathrm{Tr}(\bm{F}_m\bm{P}_m\bm{F}^\top_m) ]+ \lambda_3\lVert\bm{W}\rVert_{2,1},\nonumber \\ &{\text{\lr{s.t.}}}\bm{V},\bm{B},\bm{W}\geq 0 \nonumber \\
\end{align}
با استفاده از نُرم $\ell_{2,1} $ بر روی سطر ماتریس $\bm{W}$ خلوتی اعمال می‌کنیم و منظم‌ساز خلوتی تابع هدف نیز توسط پارامتر $\lambda_3$ کنترل می‌شود.

\section{بهینه‌سازی}\label{OPT} 
تابع \eqref{eq:BeOpt} شامل عبارت منظم‌ساز نُرم $\ell_{2,1} $ است. از آنجا که این تابع هموار نیست، همچنین وقتی متغیرهای $\bm{V}$، $\bm{B}$ و $\bm{W}$ به‌صورت ترکیبی درنظر گرفته می‌شوند، غیرمحدب\LTRfootnote{Non-convex} می‌شود. به‌عبارت دیگر، ماتریس هسین\LTRfootnote{Hessian matrix} که از مشتقات جزئی تابع هدف در درجه‌دوم بدست می‌آید، ماتریسی نیست که نیمه مثبت معین\LTRfootnote{Semi-definite} باشد. بنابراین، تابع هدف باید به گونه‌ای بهینه‌سازی شود که در هر مرحله از الگوریتم با ثابت نگه‌داشتن سه متغیر و به‌روزرسانی متغیر باقیمانده به‌صورت محدب\LTRfootnote{Convex} باشد. عبارت $\lVert\bm{W}\rVert_{2,1}$ با استفاده از $\mathrm{Tr} (\bm{W^\top DW})$ به شکلی دیگر خفیف می‌شود، که در این حالت $\bm{D}$ یک ماتریس قطری است \cite{hu2020multi}. الگوریتم مبتنی بر تکراری ویژگی‌های بروزرسانی در دسترس است. درایه $\bm{D}$ به شکل $D_{ii} = 1/(\lVert \bm{w}_i\lVert+\epsilon),(\epsilon\leftarrow0)$ است.
که $\epsilon$ برای جلوگیری از اختلال در محاسبات مسئله بکار می‌رود. به عبارت دیگر، تابع هدف \eqref{eq:BeOpt} می‌تواند به‌صورت زیر بازنویسی شود:
\begin{align}\label{eq:1stopt}
	&\min_{\bm{V},\bm{B},\bm{W}}\lVert\bm{Y}-\bm{V}\bm{B}\rVert^2_F+\lVert\bm{V}-\bm{X}\bm{W}\rVert^2_F+\lambda_1{\mathrm{Tr}\bm{({V}^\top{LV})}}\nonumber \\
	&+\lambda_2[\mathrm{Tr}(\bm{FP}\bm{F}^\top)+\sum_{m=1}^{g}\frac{n_m}{n}\mathrm{Tr}(\bm{F}_m\bm{P}_m\bm{F}^\top_m) ]+ \lambda_3(\bm{W}^\top \bm{DW}),\nonumber\\
	& {\text{\lr{s.t.}}}\bm{V},\bm{B},\bm{W}\geq 0 
\end{align}
ما برای افزودن محدودیت نامنفی در تابع، ضرایب لاگرانژی\LTRfootnote{Lagrangian multipliers} را معرفی می‌کنیم. 
$\bm{\Phi}$، $\bm{\Psi}$
و $\bm{\Omega}$ برای محدود کردن $\bm{V}$، $\bm{B}$ و $\bm{W}$ به ترتیب، جایی که $\bm{\Phi} \in\mathbb {R}^{n \times k}$, $\bm{\Psi} \in\mathbb{R}^{k \times l}$, $\bm{\Omega} \in\mathbb{R}^ {d \times k}$، در نتیجه، تابع \eqref{eq:1stopt} معادل زیر است:
\begin{align}\label{eq:2ndopt}
	&\min_{\bm{V}, \bm{B}, \bm{W}}\lVert\bm{Y}-\bm{V}\bm{B}\lVert^2_F + \lVert\bm{V}-\bm{X}\bm{W}\lVert^2_F+ \lambda_1\mathrm{Tr}(\bm{V}^\top \bm{LV})\nonumber \\
	&+\lambda_2[\mathrm{Tr}(\bm{F}\bm{P}\bm{F}^\top) + \sum_{m=1}^{g}\frac{n_m}{n}\mathrm{Tr}(\bm{F}_m\bm{P}_m\bm{F}_m^\top)]\nonumber\\
	&+\lambda_3\mathrm{Tr}(\bm{W}^\top \bm{D}\bm{W})- \mathrm{Tr}(\bm{\Phi}\bm{V}^\top)    -\mathrm{Tr}(\bm{\Psi}\bm{B}^\top)-\mathrm{Tr}(\bm{\Omega}\bm{W}^\top)\nonumber\\
\end{align}
با توجه به ماتریس $\bm{A}$, $\lVert \bm{A} \lVert^2_F = \mathrm{Tr}(\bm{A}^\top \bm{A})$. بنابراین، تابع \eqref{eq:2ndopt} تبدیل می‌شود:

\begin{align}\label{eq:3rdopt}
	C =  &\mathrm{Tr}[(\bm{Y} - \bm{V}\bm{B})^\top (\bm{Y} - \bm{V}\bm{B})]+ \mathrm{Tr}[(\bm{V}-\bm{X}\bm{W})^\top(\bm{V} - \bm{X}\bm{W})]\nonumber\\
	&+\lambda_1\mathrm{Tr}(\bm{V}^\top \bm{LV})+\lambda_2[\mathrm{Tr}(\bm{F}\bm{P}\bm{F}^\top)+ \sum_{m=1}^{g}\frac{n_m}{n}\mathrm{Tr}(\bm{F}_m\bm{P}_m\bm{F}_m^\top)]\nonumber\\
	&+ 2 \lambda_3\mathrm{Tr}(\bm{W}^\top \bm{D}\bm{W})- \mathrm{Tr}(\bm{\Phi}\bm{V}^\top) -\mathrm{Tr}(\bm{\Psi}\bm{B}^\top) -\mathrm{Tr}(\bm{\Omega}\bm{W}^\top)
\end{align}
که $\bm{F}=\bm{XWB}$ و $‎\bm{F}_m=\bm{X}_m\bm{WB}‎$, $\forall m \{1,{\dots},{g}\}$. مشتقات جزئی تابع \eqref{eq:3rdopt} با توجه به متغیرهای $‎\bm{V}$, $ \bm{B}$ و $\bm{W}$ عبارتند از:
\begin{align}\label{eq:derW}
	&\frac{\partial C}{\partial\bm{W}} = {-\bm{X}^\top \bm{V} + \bm{X}^\top \bm{XW} + \lambda_2[\bm{X}^\top \bm{XWBPB}^\top}\nonumber\\
	&+ \sum_{m=1}^{g}\frac{n_m}{n} \bm{X}^\top_m \bm{X}_m\bm{WBP}_m\bm{B}^\top]+ 2\lambda_3\bm{DW}-\bm{\Omega}
\end{align}

\begin{align}\label{eq:derB}
	\frac{\partial C}{\partial \bm{B}} = &{-\bm{V}^\top \bm{Y} + \bm{V}^\top \bm{VB} + \lambda_2[\bm{W}^\top \bm{X}^\top \bm{XWBP}} \nonumber \\
	& \quad\quad+ \sum_{m=1}^{n} \frac{n_m}{n}\bm{W}^\top \bm{X}^\top_m \bm{X}_m\bm{WBP}_m] - \bm{\Phi}
\end{align}	

\begin{align}\label{eq:derV}
	\frac{\partial C}{\partial \bm{V}} = {\bm{V} - 2\bm{XW} - 2\bm{YB}^\top + \bm{VBB}^\top + \lambda_1\bm{LV} -\bm{\Psi}}.
\end{align}

با قرار دادن مشتقات جزئی \eqref{eq:derW}، \eqref{eq:derB} و \eqref{eq:derV} برابر با صفر، می‌توانیم نقاط بهینه\LTRfootnote{Optimal points} تابع را پیدا کنیم. با توجه به شرایط\LTRfootnote{Conditions} 
\lr{Karush-Kuhn-Tucker}، $\bm{W}\odot\bm{\Omega}=\bm{0}$، $\bm{B}\odot\bm{\Phi}=\bm{0}$ و $\bm{V}\odot\bm{\Psi}=\bm{0}$ را تنظیم می‌کنیم. با حل این معادلات، توابع به‌روزرسانی زیر را برای $\bm{W}$، $\bm{B}$ و $\bm{V}$ بدست می‌آوریم:‎\\
\begin{align*}
	&\bm{\Omega}_{SGC}=\bm{X}^\top \bm{XWBCB}^\top & \\
	&\bm{\Omega}_{DGC}=\bm{X}^\top \bm{XWBAB}^\top & \\
	&\bm{\Omega}_{SLC}=\frac{n_m}{n}\bm{X}^\top_m \bm{X}_m\bm{WBC}_m\bm{B}^\top & \\
	&\bm{\Omega}_{DLC}=\frac{n_m}{n}\bm{X}^\top_m\bm{X}_m\bm{WBA}_m\bm{B}^\top & \\
\end{align*}
\begin{align}\label{eq:updW}
	\bm{W} \leftarrow \bm{W} \odot \frac{\bm{X}^\top \bm{V} + \lambda_2[\bm{\Omega}_{SGC}+ \sum_{m=1}^{g}\bm{\Omega}_{SLC}]}
	{\bm{X}^\top \bm{XW} + \lambda_2[\bm{\Omega}_{DGC} +\sum_{m=1}^{g}\bm{\Omega}_{DLC}]+\lambda_3\bm{DW}}
\end{align}
\begin{align*}
	&\bm{\Xi}_{SLC}=\frac{n_m}{n}\bm{W}^\top \bm{X}^\top_m \bm{X}_m\bm{WBC}_m & \\
	&\bm{\Xi}_{DLC}=\frac{n_m}{n}\bm{W}^\top \bm{X}^\top_m \bm{X}_m\bm{WBA}_m & \\
	&\bm{\Xi}_{SGC}=\bm{W}^\top \bm{X}^\top \bm{XWBC} & \\
	&\bm{\Xi}_{DGC}=\bm{W}^\top \bm{X}^\top \bm{XWBA} & \\
\end{align*}
\begin{align}\label{eq:updB}
	\bm{B} \leftarrow \bm{B}\odot \frac{\bm{V}^\top \bm{Y} +\lambda_2[\bm{\Xi}_{SGC} +\sum_{m=1}^{g} \bm{\Xi}_{SLC} ]}{\bm{V}^\top \bm{VB} + \lambda_2[\bm{\Xi}_{DGC} +\sum_{m=1}^{g} \bm{\Xi}_{DLC}] }
\end{align}
\begin{align}\label{eq:updV}
	\bm{V} \leftarrow \bm{V} \odot \frac{\bm{XW} + \bm{YB}^\top + \lambda_1 \bm{SV}}{\bm{V} +\bm{VBB}^\top + \lambda_1 \bm{DV}}
\end{align}

\lr{MLFS-GLOCAL} تمام ویژگی‌ها را با استفاده از $\lVert \bm{w}_i\lVert_2, (i=1,{\dots},{d})$ به ترتیب نزولی رتبه‌بندی می‌شود و به ما امکان می‌دهد تا ویژگی‌های با بالاترین امتیاز را بدست‌ آوریم. الگوریتم \ref{alg:algorithm} مدل \lr{MLFS-GLOCAL} را به‌صورت جزئیات تشریح می‌کند.

\lr{
	\begin{algorithm}[tb]	
		\setstretch{1} 
		\caption{Multi-label Feature Selection with Global and Local label correlation\\ {(MLFS-GLOCAL)}}
		\begin{flushleft}
			\textbf{Input}: Feature matrix $\bm{X}$$\in$$\mathbb{R}^ {n \times d}$ and Label matrix
			$\bm{Y}$$\in$$\mathbb{R}^{n \times c}$, Regularization parameters $\lambda_1,\lambda_2$, and $\lambda_3$, and latent factor $k$;\\
			\textbf{Output}: Feature score $s_i=\lVert\bm{w}_i\rVert, \forall i\in\{1,2,{\dots},d\}$.
		\end{flushleft}
		\begin{algorithmic}[1] %[1] enables line numbers
			\STATE \textbf{Initialize} $\bm{V},\bm{W},\bm{B}$ randomly; $t=0;$
			\WHILE{$t <$ MaxIteration}
			\STATE Update $D_{ii}\leftarrow\frac{1}{\lVert\bm{w}_i\rVert+\epsilon}$;
			\STATE  Update $\bm{W}$ by \eqref{eq:updW};
			\STATE Update $\bm{B}$ by \eqref{eq:updB};
			\STATE Update ${\bm{V}}$ by \eqref{eq:updV};
			\STATE $t = t + 1$;
			%\STATE $\textbf{Until}$ Convergence;
			\ENDWHILE
			\STATE $\textbf{Return}$ $\bm{W}$;
			\STATE Evaluate the feature score by $s_i=\lVert\bm{w}_i\rVert$.
		\end{algorithmic}
		\label{alg:algorithm}
	\end{algorithm}
